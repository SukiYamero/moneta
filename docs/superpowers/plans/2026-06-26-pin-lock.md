# PIN lock + biometric unlock — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an optional per-device lock that encrypts the cached OAuth token at rest and unlocks via biometrics (WebAuthn PRF) with a mandatory 4-digit PIN fallback.

**Architecture:** Envelope encryption — one random DEK encrypts the token once; the DEK is wrapped separately by a PBKDF2(PIN) key and by a WebAuthn-PRF-derived key. A `pinLock.ts` core (crypto + IndexedDB vault, TDD) is orchestrated by a `lockStore.ts` zustand store that wires the lock/unlock state, the Page Visibility triggers, and the handoff to `authStore`. A minimal `LockScreen` gates the app.

**Tech Stack:** TypeScript (strict), WebCrypto (AES-GCM / PBKDF2 / HKDF), WebAuthn PRF, dexie (IndexedDB), zustand, React 19, Vitest + Testing Library + `fake-indexeddb`.

```ts
import { db, VAULT_ID, type LockVault } from '@/lib/db'
import type { AuthSession } from '@/lib/auth'

const PIN_ITERATIONS = 310_000
const enc = new TextEncoder()
const dec = new TextDecoder()

export class WrongPinError extends Error {
  constructor() {
    super('lock: wrong pin')
    this.name = 'WrongPinError'
  }
}

function randomBytes(length: number): Uint8Array {
  return crypto.getRandomValues(new Uint8Array(length))
}

function generateDek(): Uint8Array {
  return randomBytes(32)
}

function importAesKey(raw: Uint8Array): Promise<CryptoKey> {
  return crypto.subtle.importKey('raw', raw, 'AES-GCM', false, ['encrypt', 'decrypt'])
}

async function aesEncrypt(
  key: CryptoKey,
  data: Uint8Array,
): Promise<{ iv: Uint8Array; cipher: Uint8Array }> {
  const iv = randomBytes(12)
  const cipher = await crypto.subtle.encrypt({ name: 'AES-GCM', iv }, key, data)
  return { iv, cipher: new Uint8Array(cipher) }
}

async function aesDecrypt(key: CryptoKey, iv: Uint8Array, cipher: Uint8Array): Promise<Uint8Array> {
  const plain = await crypto.subtle.decrypt({ name: 'AES-GCM', iv }, key, cipher)
  return new Uint8Array(plain)
}

async function derivePinKey(pin: string, salt: Uint8Array, iterations: number): Promise<CryptoKey> {
  const base = await crypto.subtle.importKey('raw', enc.encode(pin), 'PBKDF2', false, ['deriveKey'])
  return crypto.subtle.deriveKey(
    { name: 'PBKDF2', salt, iterations, hash: 'SHA-256' },
    base,
    { name: 'AES-GCM', length: 256 },
    false,
    ['encrypt', 'decrypt'],
  )
}

export async function hasVault(): Promise<boolean> {
  return (await db.vault.get(VAULT_ID)) !== undefined
}

export async function enableLock(opts: { pin: string; session: AuthSession }): Promise<void> {
  const dek = generateDek()
  const dekKey = await importAesKey(dek)
  const token = await aesEncrypt(dekKey, enc.encode(JSON.stringify(opts.session)))

  const pinSalt = randomBytes(16)
  const pinKey = await derivePinKey(opts.pin, pinSalt, PIN_ITERATIONS)
  const pinWrap = await aesEncrypt(pinKey, dek)

  const vault: LockVault = {
    schemaVersion: 1,
    tokenCipher: token.cipher,
    tokenIv: token.iv,
    pinSalt,
    pinIterations: PIN_ITERATIONS,
    dekWrappedByPin: pinWrap.cipher,
    pinWrapIv: pinWrap.iv,
    failedAttempts: 0,
    lastActiveAt: Date.now(),
  }
  await db.vault.put({ id: VAULT_ID, ...vault })
}

async function readVault(): Promise<LockVault> {
  const vault = await db.vault.get(VAULT_ID)
  if (!vault) throw new Error('lock: no vault')
  return vault
}

async function decryptSession(vault: LockVault, dek: Uint8Array): Promise<AuthSession> {
  const dekKey = await importAesKey(dek)
  const plain = await aesDecrypt(dekKey, vault.tokenIv, vault.tokenCipher)
  return JSON.parse(dec.decode(plain)) as AuthSession
}

export async function unlockWithPin(pin: string): Promise<AuthSession> {
  const vault = await readVault()
  const pinKey = await derivePinKey(pin, vault.pinSalt, vault.pinIterations)
  let dek: Uint8Array
  try {
    dek = await aesDecrypt(pinKey, vault.pinWrapIv, vault.dekWrappedByPin)
  } catch {
    throw new WrongPinError()
  }
  return decryptSession(vault, dek)
}
```

```ts
export async function unlockWithPin(pin: string): Promise<AuthSession> {
  const vault = await readVault()
  if (vault.failedAttempts >= MAX_ATTEMPTS) throw new LockedOutError()

  const pinKey = await derivePinKey(pin, vault.pinSalt, vault.pinIterations)

  return decryptSession(vault, dek)
}
```

Add the PRF/HKDF helpers and the WebAuthn ceremonies:

```ts
const HKDF_INFO = enc.encode('moneta-lock-dek')

export async function isBiometricAvailable(): Promise<boolean> {
  const api = globalThis.PublicKeyCredential
  if (!api?.isUserVerifyingPlatformAuthenticatorAvailable) return false
  try {
    return await api.isUserVerifyingPlatformAuthenticatorAvailable()
  } catch {
    return false
  }
}

async function deriveBiometricKey(prfSecret: ArrayBuffer): Promise<CryptoKey> {
  const base = await crypto.subtle.importKey('raw', prfSecret, 'HKDF', false, ['deriveKey'])
  return crypto.subtle.deriveKey(
    { name: 'HKDF', hash: 'SHA-256', salt: new Uint8Array(0), info: HKDF_INFO },
    base,
    { name: 'AES-GCM', length: 256 },
    false,
    ['encrypt', 'decrypt'],
  )
}

function prfResult(credential: PublicKeyCredential): ArrayBuffer | undefined {
  const ext = credential.getClientExtensionResults() as {
    prf?: { results?: { first?: ArrayBuffer } }
  }
  return ext.prf?.results?.first
}

async function registerBiometric(
  prfSalt: Uint8Array,
): Promise<{ credentialId: Uint8Array; secret: ArrayBuffer } | null> {
  const created = (await navigator.credentials.create({
    publicKey: {
      challenge: randomBytes(32),
      rp: { name: 'Moneta', id: location.hostname },
      user: { id: randomBytes(16), name: 'moneta-lock', displayName: 'Moneta lock' },
      pubKeyCredParams: [
        { type: 'public-key', alg: -7 },
        { type: 'public-key', alg: -257 },
      ],
      authenticatorSelection: {
        authenticatorAttachment: 'platform',
        userVerification: 'required',
        residentKey: 'required',
      },
      extensions: { prf: { eval: { first: prfSalt } } },
    },
  })) as PublicKeyCredential | null
  if (!created) return null

  const credentialId = new Uint8Array(created.rawId)
  const secret = await evaluatePrf(credentialId, prfSalt)
  return secret ? { credentialId, secret } : null
}

async function evaluatePrf(
  credentialId: Uint8Array,
  prfSalt: Uint8Array,
): Promise<ArrayBuffer | undefined> {
  const assertion = (await navigator.credentials.get({
    publicKey: {
      challenge: randomBytes(32),
      allowCredentials: [{ type: 'public-key', id: credentialId }],
      userVerification: 'required',
      extensions: { prf: { eval: { first: prfSalt } } },
    },
  })) as PublicKeyCredential | null
  return assertion ? prfResult(assertion) : undefined
}

export async function biometricEnabled(): Promise<boolean> {
  const vault = await db.vault.get(VAULT_ID)
  return vault?.biometric !== undefined
}

export async function unlockWithBiometric(): Promise<AuthSession> {
  const vault = await readVault()
  if (!vault.biometric) throw new BiometricUnavailableError()
  const secret = await evaluatePrf(vault.biometric.credentialId, vault.biometric.prfSalt)
  if (!secret) throw new BiometricUnavailableError()

  const bioKey = await deriveBiometricKey(secret)
  const dek = await aesDecrypt(bioKey, vault.biometric.prfWrapIv, vault.biometric.dekWrappedByPrf)
  activeDek = dek
  return decryptSession(vault, dek)
}
```

```ts
export async function enableLock(opts: {
  pin: string

  biometric?: boolean

  const dek = generateDek()
  const dekKey = await importAesKey(dek)
  const token = await aesEncrypt(dekKey, enc.encode(JSON.stringify(opts.session)))

  const pinSalt = randomBytes(16)
  const pinKey = await derivePinKey(opts.pin, pinSalt, PIN_ITERATIONS)
  const pinWrap = await aesEncrypt(pinKey, dek)

  const vault: LockVault = {
    schemaVersion: 1,
    tokenCipher: token.cipher,
    tokenIv: token.iv,
    pinSalt,
    pinIterations: PIN_ITERATIONS,
    dekWrappedByPin: pinWrap.cipher,
    pinWrapIv: pinWrap.iv,
    failedAttempts: 0,
    lastActiveAt: Date.now(),
  }

  if (opts.biometric) {
    const prfSalt = randomBytes(32)
    const registered = await registerBiometric(prfSalt)
    if (registered) {
      const bioKey = await deriveBiometricKey(registered.secret)
      const bioWrap = await aesEncrypt(bioKey, dek)
      vault.biometric = {
        credentialId: registered.credentialId,
        prfSalt,
        dekWrappedByPrf: bioWrap.cipher,
        prfWrapIv: bioWrap.iv,
      }
    }
  }

  await db.vault.put({ id: VAULT_ID, ...vault })
}
```

- [ ] **Step 7: Implement `lockStore.ts`**

```ts
import { create } from 'zustand'
import { useAuthStore } from '@/lib/authStore'
import {
  enableLock,
  hasVault,
  isBackgroundExpired,
  isBiometricAvailable,
  LockedOutError,
  markActive,
  resetVault,
  unlockWithBiometric,
  unlockWithPin,
} from '@/lib/pinLock'
import type { AuthSession } from '@/lib/auth'

type LockPhase = 'unknown' | 'unlocked' | 'locked'

type LockState = {
  phase: LockPhase
  biometricAvailable: boolean
  error: string | null
  init: () => Promise<void>
  enable: (pin: string, biometric: boolean) => Promise<void>
  unlockPin: (pin: string) => Promise<void>
  unlockBiometric: () => Promise<void>
  onHidden: () => void
  onVisible: () => Promise<void>
  reset: () => Promise<void>
}

async function resume(set: (s: Partial<LockState>) => void, unlock: () => Promise<AuthSession>) {
  try {
    const session = await unlock()
    await useAuthStore.getState().hydrate(session)
    set({ phase: 'unlocked', error: null })
  } catch (e) {
    if (e instanceof LockedOutError) {
      await resetVault()
      useAuthStore.getState().logout()
      set({ phase: 'locked', error: 'locked out' })
      return
    }
    set({ error: e instanceof Error ? e.message : 'unlock failed' })
  }
}

export const useLockStore = create<LockState>((set, get) => ({
  phase: 'unknown',
  biometricAvailable: false,
  error: null,
  init: async () => {
    const [locked, biometricAvailable] = await Promise.all([hasVault(), isBiometricAvailable()])
    set({ phase: locked ? 'locked' : 'unlocked', biometricAvailable })
  },
  enable: async (pin, biometric) => {
    const session = useAuthStore.getState().session
    if (!session) throw new Error('lock: no session to protect')
    await enableLock({ pin, session, biometric })
    set({ phase: 'unlocked' })
  },
  unlockPin: (pin) => resume(set, () => unlockWithPin(pin)),
  unlockBiometric: () => resume(set, () => unlockWithBiometric()),
  onHidden: () => {
    if (get().phase === 'unlocked') void markActive()
  },
  onVisible: async () => {
    if (get().phase === 'unlocked' && (await isBackgroundExpired())) set({ phase: 'locked' })
  },
  reset: async () => {
    await resetVault()
    useAuthStore.getState().logout()
    set({ phase: 'locked', error: null })
  },
}))
```
